🎯 Sprint 5.1: Supabase 데이터베이스 마이그레이션 및 RLS 구축

PostgreSQL 기반의 Supabase 환경에서 안전하게 멀티 테넌시를 유지하고 인프라 오염을 막기 위한 최적의 엔터프라이즈 스키마 및 RLS(Row Level Security) 스크립트입니다.

📄 구현 파일 1: database/migrations/02_init_text_schema.sql

사용자 프로필, 마크다운 기반 챌린지 맵, 그리고 매주 조제되는 4단 처방전 데이터가 들어가는 테이블 구조 정의서입니다.
sql
-- 1. UUID 및 외래키 연동용 유저 프로필 테이블 생성
CREATE TABLE public.user_profiles (
    id UUID REFERENCES auth.users ON DELETE CASCADE PRIMARY KEY,
    nickname TEXT NOT NULL UNIQUE,
    persona_type TEXT DEFAULT 'easy_peasy',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT TIMEZONE('utc'::text, NOW()) NOT NULL
);

-- 2. 마크다운에서 추출된 4단 처방전 결과 데이터 적재 테이블 생성
CREATE TABLE public.prescription_logs (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    user_id UUID REFERENCES public.user_profiles(id) ON DELETE CASCADE NOT NULL,
    original_target TEXT NOT NULL,
    five_year_old_version TEXT NOT NULL,
    cheat_key_action TEXT NOT NULL,
    phonics_sound TEXT NOT NULL,
    is_stolen_from_peer BOOLEAN DEFAULT FALSE, -- 타인의 표현을 스틸했는지 여부 방어용 플래그
    source_log_id BIGINT REFERENCES public.prescription_logs(id) ON DELETE SET NULL, -- 원본 출처 추적
    created_at TIMESTAMP WITH TIME ZONE DEFAULT TIMEZONE('utc'::text, NOW()) NOT NULL
);

-- 인덱스 설정을 통한 유저별 조회 속도 최적화 레이어 가동
CREATE INDEX idx_prescription_logs_user_id ON public.prescription_logs(user_id);

📄 구현 파일 2: database/migrations/05_rls_theft_policy.sql

기본적으로 자신의 데이터만 수정할 수 있지만, 다른 동료 유저의 4단 처방 처방전 표현을 읽고 내 계정으로 안전하게 퍼올 수(Steal) 있도록 허용하는 특화 Row Level Security 정책 스크립트입니다.
sql
-- 1. 테이블 단위 Row Level Security 보안 락 가동
ALTER TABLE public.user_profiles ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.prescription_logs ENABLE ROW LEVEL SECURITY;

-- 2. 프로필용 RLS: 본인의 프로필만 수정 가능, 읽기는 전체 공개 허용
CREATE POLICY "전체 유저는 타인의 닉네임을 조회할 수 있다" 
ON public.user_profiles FOR SELECT USING (true);

CREATE POLICY "인증된 유저는 본인의 프로필만 생성 및 수정할 수 있다" 
ON public.user_profiles FOR ALL USING (auth.uid() = id);

-- 3. 처방전 및 표현 스틸용 RLS 가드레일 수립
-- 규칙 A: 다른 유저의 처방전 데이터를 자유롭게 SELECT(조회)하여 볼 수 있게 열어둠 (합법적 표현 스틸의 전제 조건)
CREATE POLICY "모든 인증된 유저는 동료의 처방전을 읽을 수 있다"
ON public.prescription_logs FOR SELECT
TO authenticated
USING (true);

-- 규칙 B: 데이터의 삽입(INSERT)은 오직 본인의 auth.uid() 명의로만 수행 가능하도록 강제하여 타인 명의 도용 차단
CREATE POLICY "유저는 오직 본인의 명의로만 처방전이나 스틸 데이터를 삽입할 수 있다"
ON public.prescription_logs FOR INSERT
TO authenticated
WITH CHECK (auth.uid() = user_id);

-- 규칙 C: 수정(UPDATE) 및 삭제(DELETE)는 철저히 본인이 소유한 데이터 행만 가능하도록 격리 가드레일 설치
CREATE POLICY "유저는 본인이 생성한 처방전 데이터만 수정 및 삭제할 수 있다"
ON public.prescription_logs FOR UPDATE
TO authenticated
USING (auth.uid() = user_id)
WITH CHECK (auth.uid() = user_id);
Use code with caution.
