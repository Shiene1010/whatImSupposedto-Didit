이 모듈은 백엔드 파이프라인 연동 중 발생하는 치명적 예외를 원천 방어합니다. 데이터베이스의 원자성(Atomicity)을 보장하기 위해 제한 시간 내에 작업이 완료되지 않으면 클라이언트 측과 데이터베이스 측의 상태를 동시에 완전 무효화합니다.

📄 1. PostgreSQL 원격 복구 저장 프로시저: database/migrations/08_rollback_rpc.sql

데이터 트랜잭션 충돌 시 안전한 상태로 원상 복구하기 위해 Supabase(PostgreSQL) 레이어에 직접 적재하는 원격 트랜잭션 안전 제어 함수입니다.
sql
CREATE OR REPLACE FUNCTION public.execute_safe_prescription_transaction(
    p_user_id UUID,
    p_original_target TEXT,
    p_five_year_old TEXT,
    p_cheat_key TEXT,
    p_phonics_sound TEXT
)
RETURNS BOOLEAN AS $$
BEGIN
    -- 1. 데이터 원자성 보장을 위한 내부 격리 트랜잭션 블록 개시
    -- 충돌 발생 시 이 블록 내부의 모든 INSERT/UPDATE는 원천 무효화됨
    INSERT INTO public.prescription_logs (
        user_id, 
        original_target, 
        five_year_old_version, 
        cheat_key_action, 
        phonics_sound
    ) VALUES (
        p_user_id, 
        p_original_target, 
        p_five_year_old, 
        p_cheat_key, 
        p_phonics_sound
    );

    RETURN TRUE;

EXCEPTION WHEN OTHERS THEN
    -- 내부 구문 파괴나 락 충돌 등 예외 감지 시 즉각 예외를 상위로 던져 트랜잭션 원천 롤백
    RAISE EXCEPTION 'DATABASE_TRANSACTION_CRASH_ROLLED_BACK';
    RETURN FALSE;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

📄 2. 백엔드 코어 롤백 핸들러: src/operation/rollbackHandler.ts

Node.js/TypeScript 환경에서 8초 타이머 데드라인을 강제 집행하고 데이터베이스 RPC를 안전하게 래핑하는 코어 프로세스 모듈입니다.
typescript
interface TransactionPayload {
  userId: string;
  originalTarget: string;
  fiveYearOldVersion: string;
  cheatKeyAction: string;
  phonicsSound: string;
}

/**
 * RollbackHandler
 * 8초 데드라인 타이머를 가동하여 마크다운 파싱 및 DB 유효성 검증 실패 시 상태를 완전 복구합니다.
 */
export class RollbackHandler {
  private supabaseClient: any;
  private deadlineMs = 8000; // 8초 하드코딩 데드라인 임계치 가드레일

  constructor(supabaseClient: any) {
    this.supabaseClient = supabaseClient;
  }

  /**
   * executeWith8SecRollback Guard
   * 8초 데드라인 내에 완료되지 않으면 프로미스를 강제 거부(Reject)하고 롤백 컨텍스트를 반환합니다.
   */
  public async executeWith8SecRollback(payload: TransactionPayload): Promise<{ success: boolean; status: string }> {
    
    // 1. 8초 타이머 거부(Reject) 프로미스 생성
    const rollbackTimerPromise = new Promise<never>((_, reject) => {
      setTimeout(() => {
        reject(new Error('TRANSACTION_TIMEOUT_8SEC_LIMIT_REACHED'));
      }, this.deadlineMs);
    });

    // 2. 실제 Supabase PostgreSQL RPC 안전 트랜잭션 가동 프로미스
    const dbTransactionPromise = async () => {
      const { data, error } = await this.supabaseClient.rpc('execute_safe_prescription_transaction', {
        p_user_id: payload.userId,
        p_original_target: payload.originalTarget,
        p_five_year_old: payload.fiveYearOldVersion,
        p_cheat_key: payload.cheatKeyAction,
        p_phonics_sound: payload.phonicsSound
      });

      if (error) {
        throw new Error(`DB_RPC_FAILED: ${error.message}`);
      }
      return data;
    };

    try {
      // 3. 8초 타이머와 실제 처리 프로미스의 실시간 레이스 경쟁 유도
      await Promise.race([dbTransactionPromise(), rollbackTimerPromise]);
      
      console.log(`🔵 [ROLLBACK GUARD] 유저 [${payload.userId}] 트랜잭션 8초 이내에 정상 처리 완료.`);
      return { success: true, status: 'TRANSACTION_COMMITTED' };

    } catch (error: any) {
      console.error(`🚨 [CRITICAL ROLLBACK TRIGGERED] 사유: ${error.message}`);
      
      // 8초 타임아웃 또는 데이터 파괴 예외 발생 시 상위 아키텍처에 롤백 상태 전파
      // PostgreSQL RPC 특성상 에러가 터지거나 중단되면 자산은 자동으로 원상 복구(Rollback) 상태 유지
      return { 
        success: false, 
        status: error.message === 'TRANSACTION_TIMEOUT_8SEC_LIMIT_REACHED' 
          ? 'TIMEOUT_ROLLED_BACK' 
          : 'EXCEPTION_ROLLED_BACK' 
      };
    }
  }
}

// Supabase RPC 처리가 내부 장애로 인해 12,000ms 동안 블로킹 상태라고 에뮬레이션
const mockStuckSupabase = {

};

const handler = new RollbackHandler(mockStuckSupabase);

const samplePayload = {

};

// 가동 및 차단 테스트
const transactionReport = await handler.executeWith8SecRollback(samplePayload);
